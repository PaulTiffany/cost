\begin{figure}[!htb]
\centering
\resizebox{0.85\linewidth}{!}{%
\begin{tikzpicture}[
    font=\normalsize,
    node distance=8mm and 16mm,
    box/.style={draw, rounded corners=2pt, minimum height=10mm, minimum width=44mm, align=center, inner sep=4pt, fill=white, draw=black!60},
    inputbox/.style={box, fill=gray!8},
    computebox/.style={box, fill=blue!6, draw=blue!50},
    decisionbox/.style={box, fill=orange!10, draw=orange!60, minimum width=72mm, minimum height=14mm},
    routebox/.style={box, minimum width=32mm, minimum height=10mm},
    oneshot/.style={routebox, fill=green!12, draw=green!60!black},
    staged/.style={routebox, fill=blue!12, draw=blue!60!black},
    decomp/.style={routebox, fill=purple!10, draw=purple!60!black},
    arr/.style={->, thick, >=stealth},
    cond/.style={font=\footnotesize, color=black!75, align=left, anchor=west},
]
    % Vertical layered layout
    \node[inputbox] (inp) at (0, 0) {\textbf{Input:} prompt $P$, budget $T$, encoder $\Phi$};
    \node[computebox, below=of inp] (parse) {\textbf{Parse} constraints $C_1, \ldots, C_k$ from $P$};

    % Parallel pair: regime index + budget estimate
    \node[computebox, below left=12mm and 4mm of parse] (rho) {\textbf{Compute} $\hat{\rho} = \max_{i<j}\cos(\Phi(C_i),\Phi(C_j))$};
    \node[computebox, below right=12mm and 4mm of parse] (budget) {\textbf{Estimate} $\delta = \hat{L} \cdot T$};

    % Converge to screened cost
    \node[computebox, below=18mm of parse] (cost) {\textbf{Screened cost} $\widehat{\delta}_{\min} = (\tau/m)\sqrt{k/(1{-}\hat{\rho}(k{-}1))}$};

    % Decision node (wider)
    \node[decisionbox, below=of cost] (route) {\textbf{Route} on $\hat{\rho}$ and $\delta/\widehat{\delta}_{\min}$};

    % Three outputs
    \node[oneshot, below left=14mm and 26mm of route] (oneshot) {\textsc{One-Shot}};
    \node[staged, below=14mm of route] (staged) {\textsc{Staged}};
    \node[decomp, below right=14mm and 26mm of route] (decomp) {\textsc{Decompose-First}};

    % Arrows
    \draw[arr] (inp) -- (parse);
    \draw[arr] (parse.south) -- (rho.north);
    \draw[arr] (parse.south) -- (budget.north);
    \draw[arr] (rho.south) -- (cost.north);
    \draw[arr] (budget.south) -- (cost.north);
    \draw[arr] (cost) -- (route);
    \draw[arr] (route.south) -- (oneshot.north);
    \draw[arr] (route.south) -- (staged.north);
    \draw[arr] (route.south) -- (decomp.north);

    % Decision conditions on the route arrows
    \node[cond, color=green!50!black] at ($(route.south)+(-3.6, -0.7)$) {$\hat{\rho}<0.15$ \\ or $\delta/\widehat{\delta}_{\min}>2$};
    \node[cond, color=blue!50!black] at ($(route.south)+(-0.4, -0.7)$) {$\hat{\rho}<0.5$ \\ and $\delta/\widehat{\delta}_{\min}>1.2$};
    \node[cond, color=purple!50!black] at ($(route.south)+(2.7, -0.7)$) {else: \\ $\hat{\rho}\ge 0.5$};
\end{tikzpicture}
}
\caption{Algorithm~\ref{alg:routing} as a decision tree (schematic). The screening statistic $\hat{\rho}$ and budget ratio $\delta/\widehat{\delta}_{\min}$ are computed from the prompt before generation, then composed at a single routing junction with three terminal regimes. The conditions printed beside each terminal are exactly those in lines 6--12 of Algorithm~\ref{alg:routing}. Provenance: \texttt{ci/illustration\_lineage.json}; draft via gpt-5.4-image-2; final asset is hand-authored TikZ. Schematic illustration only; not empirical evidence.}
\label{fig:algorithm1_routing}
\end{figure}
